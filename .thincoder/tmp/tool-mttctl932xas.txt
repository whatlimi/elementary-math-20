  1641	/* 高亮命中词(支持多词，不嵌套mark)：遍历 .pat-hd-t/.pat-hd-d 文本节点，对命中的词包 <mark> */
  1642	function hlTerms(sec,terms){
  1643	  const tgt=[...sec.querySelectorAll('.pat-hd-t,.pat-hd-d')];
  1644	  tgt.forEach(el=>{
  1645	    if(!el) return;
  1646	    const walker=document.createTreeWalker(el,NodeFilter.SHOW_TEXT,null);
  1647	    const nodes=[]; let n;
  1648	    while((n=walker.nextNode())){
  1649	      // 跳过已处于 mark 内的文本节点(避免嵌套)
  1650	      let p=n.parentNode, inside=false;
  1651	      while(p&&p!==el){ if(p.nodeName==='MARK'){inside=true;break;} p=p.parentNode; }
  1652	      if(!inside) nodes.push(n);
  1653	    }
  1654	    nodes.forEach(tn=>{
  1655	      const low=tn.nodeValue.toLowerCase();
  1656	      // 收集该文本节点命中词集合(按出现位置)
  1657	      const hits=[];
  1658	      terms.forEach(t=>{ if(low.includes(t)) hits.push(t); });
  1659	      if(!hits.length) return;
  1660	      let frag; const text=tn.nodeValue;
  1661	      // 简化：仅当文本含某词时在该词外包mark，多词逐词处理易重叠 → 用一个顺序扫描
  1662	      frag=document.createDocumentFragment(); let pos=0; const used=new Set();
  1663	      // 对每个位置找最早出现的任一命中词
  1664	      while(pos<text.length){
  1665	        let best=null, bp=-1;
  1666	        terms.forEach(t=>{ const li=low.indexOf(t,pos); if(li!==-1&&(bp===-1||li<bp)){bp=li;best=t;} });
  1667	        if(best===null||bp===-1){ frag.appendChild(document.createTextNode(text.slice(pos))); pos=text.length; break; }
  1668	        if(bp>pos) frag.appendChild(document.createTextNode(text.slice(pos,bp)));
  1669	        const mk=document.createElement('mark'); mk.className='hl'; mk.textContent=text.slice(bp,bp+best.length); frag.appendChild(mk);
  1670	        pos=bp+best.length;
  1671	      }
  1672	      tn.parentNode.replaceChild(frag,tn);
  1673	    });
  1674	  });
  1675	}
